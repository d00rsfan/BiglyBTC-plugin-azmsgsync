cmake_minimum_required(VERSION 3.25)
project(BiglyBTC-plugin-azmsgsync VERSION 0.1.0 LANGUAGES CXX)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Toolchain.cmake")
  include(cmake/Toolchain.cmake)
else()
  include(../../cmake/Toolchain.cmake)
endif()
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL 3.5.5 EXACT REQUIRED COMPONENTS Crypto)
add_library(biglybtc_msgsync_envelope STATIC envelope.cpp)
add_library(biglybtc_msgsync_sts STATIC legacy_sts.cpp)
target_include_directories(biglybtc_msgsync_sts PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(biglybtc_msgsync_sts PRIVATE OpenSSL::Crypto)
target_compile_features(biglybtc_msgsync_sts PUBLIC cxx_std_23)
target_compile_options(biglybtc_msgsync_sts PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(biglybtc_msgsync_envelope PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(biglybtc_msgsync_envelope PRIVATE OpenSSL::Crypto)
target_compile_features(biglybtc_msgsync_envelope PUBLIC cxx_std_23)
target_compile_options(biglybtc_msgsync_envelope PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(biglybtc-msgsync-envelope-driver envelope_driver.cpp)
target_link_libraries(biglybtc-msgsync-envelope-driver PRIVATE biglybtc_msgsync_envelope)
target_compile_options(biglybtc-msgsync-envelope-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME original-msgsync-envelopes COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/fixture_test.py" "$<TARGET_FILE:biglybtc-msgsync-envelope-driver>")
set_tests_properties(original-msgsync-envelopes PROPERTIES TIMEOUT 90 LABELS "unit;fixtures")
add_executable(biglybtc-msgsync-envelope-bounds-tests bounds_test.cpp)
target_link_libraries(biglybtc-msgsync-envelope-bounds-tests PRIVATE biglybtc_msgsync_envelope)
target_compile_options(biglybtc-msgsync-envelope-bounds-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bounded-msgsync-envelopes COMMAND biglybtc-msgsync-envelope-bounds-tests)
set_tests_properties(bounded-msgsync-envelopes PROPERTIES TIMEOUT 30 LABELS "unit")
add_executable(biglybtc-msgsync-sts-driver sts_driver.cpp)
target_link_libraries(biglybtc-msgsync-sts-driver PRIVATE biglybtc_msgsync_sts biglybtc_msgsync_envelope OpenSSL::Crypto)
target_compile_options(biglybtc-msgsync-sts-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME original-msgsync-sts COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/sts_fixture_test.py" "$<TARGET_FILE:biglybtc-msgsync-sts-driver>")
set_tests_properties(original-msgsync-sts PROPERTIES TIMEOUT 180 LABELS "unit;fixtures")
add_executable(biglybtc-msgsync-sts-bounds-tests sts_bounds_test.cpp)
target_link_libraries(biglybtc-msgsync-sts-bounds-tests PRIVATE biglybtc_msgsync_sts)
target_compile_options(biglybtc-msgsync-sts-bounds-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bounded-msgsync-sts COMMAND biglybtc-msgsync-sts-bounds-tests)
set_tests_properties(bounded-msgsync-sts PROPERTIES TIMEOUT 30 LABELS "unit")
